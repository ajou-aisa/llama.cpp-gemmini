cmake_minimum_required(VERSION 3.16)
project(potal-a4-probes LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(GEMMINI_CONFIG "${REPO}/build-potal-lab/generated" CACHE PATH "DIM32 generated parameter directory")
set(LLAMA_LIB_DIR "" CACHE PATH "Existing CPU llama library directory")
set(IM2P_SIM_ROOT "" CACHE PATH "Optional real numerical simulator source root")
set(IM2P_SELECTED_GENERATION "" CACHE PATH "Verified A4/W4/DIM32 selected generation")
if(IM2P_SELECTED_GENERATION)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${IM2P_SIM_ROOT}/scripts/real_lib_cache.py" verify
        --manifest "${IM2P_SELECTED_GENERATION}/real-lib.json"
        --expected-identity a4-w4-d32 --expected-implementation GEMMINI_HP1
        --expected-block-size 32 --expected-platform "${CMAKE_SYSTEM_NAME}"
        --expected-platform-release "${CMAKE_SYSTEM_VERSION}" --expected-arch "${CMAKE_SYSTEM_PROCESSOR}"
        --artifact-kind selected RESULT_VARIABLE IM2P_VERIFY_STATUS)
    if(NOT IM2P_VERIFY_STATUS EQUAL 0)
        message(FATAL_ERROR "Matched IM2P selected artifact verification failed")
    endif()
    find_package(Threads REQUIRED)
    add_executable(potal-im2p-raw-probe im2p_raw_probe.cpp)
    target_compile_features(potal-im2p-raw-probe PRIVATE cxx_std_17)
    target_include_directories(potal-im2p-raw-probe PRIVATE "${IM2P_SIM_ROOT}/sim/include")
    target_link_libraries(potal-im2p-raw-probe PRIVATE "${IM2P_SELECTED_GENERATION}/libim2p_sim.a" Threads::Threads ${CMAKE_DL_LIBS})
endif()
set(RMD "${REPO}/ggml/src/ggml-gemmini/residual/rmd")
add_executable(potal-native-probe native_probe.cpp "${RMD}/rmd-builder.cpp" "${RMD}/rmd-bitmap-builder.cpp")
target_compile_features(potal-native-probe PRIVATE cxx_std_17)
target_compile_definitions(potal-native-probe PRIVATE GGML_GEMMINI_ACTIVATION_BITS=4 GGML_GEMMINI_WEIGHT_BITS=4 LOG_CYCLE=0)
target_include_directories(potal-native-probe PRIVATE "${GEMMINI_CONFIG}" "${REPO}/ggml/src")
add_executable(potal-native-model native_model.cpp)
target_compile_features(potal-native-model PRIVATE cxx_std_17)
target_include_directories(potal-native-model PRIVATE "${REPO}/include" "${REPO}/ggml/include")
target_link_directories(potal-native-model PRIVATE "${LLAMA_LIB_DIR}")
target_link_libraries(potal-native-model PRIVATE llama ggml ggml-base)
set_target_properties(potal-native-model PROPERTIES BUILD_RPATH "${LLAMA_LIB_DIR}")
add_library(potal-native-replay SHARED native_model.cpp)
target_compile_features(potal-native-replay PRIVATE cxx_std_17)
target_compile_definitions(potal-native-replay PRIVATE POTAL_NATIVE_LIBRARY=1)
target_include_directories(potal-native-replay PRIVATE "${REPO}/include" "${REPO}/ggml/include")
target_link_directories(potal-native-replay PRIVATE "${LLAMA_LIB_DIR}")
target_link_libraries(potal-native-replay PRIVATE llama ggml ggml-base)
set_target_properties(potal-native-replay PROPERTIES BUILD_RPATH "${LLAMA_LIB_DIR}")
enable_testing()
if(TARGET potal-im2p-raw-probe)
    add_test(NAME potal-im2p-real-raw COMMAND potal-im2p-raw-probe)
    set_tests_properties(potal-im2p-real-raw PROPERTIES TIMEOUT 120)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME potal-arithmetic-reference COMMAND "${Python3_EXECUTABLE}" "${REPO}/tests/test-potal-a4-reference.py")
foreach(LOG IN ITEMS 0 1)
    add_executable(bitmap-log${LOG} "${REPO}/tests/test-gemmini-rmd-bitmap.cpp" "${RMD}/rmd-builder.cpp" "${RMD}/rmd-bitmap-builder.cpp")
    target_compile_features(bitmap-log${LOG} PRIVATE cxx_std_17)
    target_compile_definitions(bitmap-log${LOG} PRIVATE LOG_CYCLE=${LOG})
    target_include_directories(bitmap-log${LOG} PRIVATE "${GEMMINI_CONFIG}" "${REPO}/ggml/src")
    add_test(NAME bitmap-log${LOG} COMMAND bitmap-log${LOG})
endforeach()
add_test(NAME native-direct-sparse COMMAND potal-native-probe direct sparse 32 64)
add_test(NAME native-clipped-dense COMMAND potal-native-probe clipped dense 32 64)
