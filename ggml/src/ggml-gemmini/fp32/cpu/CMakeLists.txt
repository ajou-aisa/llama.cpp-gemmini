cmake_minimum_required(VERSION 3.21)
project(fp32_cpu_reference LANGUAGES C CXX)
get_filename_component(LLAMA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_GEMMINI OFF CACHE BOOL "" FORCE)
set(GGML_GEMMINI_FP32 OFF CACHE BOOL "" FORCE)
set(GGML_POTAL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_ROOT}" llama)
add_executable(fp32-cpu-runner ../tests/runner.cpp)
target_link_libraries(fp32-cpu-runner PRIVATE llama ggml-cpu)
target_compile_features(fp32-cpu-runner PRIVATE cxx_std_17)
